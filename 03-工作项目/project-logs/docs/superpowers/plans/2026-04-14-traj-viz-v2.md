> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the v1 flat-trajectory viewer with a conversation-first v2 system: new multi-turn .traj schema, JSONL converter, updated FastAPI backend, and Vue 3 frontend.

**Architecture:** Raw Claude Code JSONL logs are converted offline by `converter_v2.py` into v2 `.traj` files (multi-turn grouped: user messages + agent runs with structured steps). The FastAPI backend reads these into an in-memory index. The Vue 3 SPA renders a conversation-first timeline.

**Tech Stack:** Python 3.10+ (FastAPI, Pydantic), Vue 3 + TypeScript + Vite, pytest + httpx for backend tests.

```python
"""Phase tagging and marker detection for v2 trajectory steps.

V2 steps use structured action objects: {"tool_name": "Read", "args": {...}}
instead of v1 flat strings like 'Read(file_path="/a.py")'.
"""
from __future__ import annotations

PHASE_NAMES = {
    "setup": ("Task Setup", "任务注入"),
    "localization": ("Localization", "问题定位"),
    "editing": ("Editing", "补丁生成"),
    "verification": ("Verification", "验证与测试"),
    "submission": ("Submission", "提交"),
}

LOCALIZATION_TOOLS = {"Read", "Glob", "Grep", "LSP", "Agent"}
EDITING_TOOLS = {"Write", "Edit", "NotebookEdit"}
VERIFICATION_PATTERNS = ["pytest", "test", "npm test", "cargo test"]
SUBMISSION_PATTERNS = ["git commit", "git push"]


def classify_action_v2(step: dict) -> str:
    """Classify a v2 step into a phase label based on its structured action."""
    action = step.get("action", {})
    tool_name = action.get("tool_name", "")

    if tool_name == "respond_to_user":
        return "inherited"

    if tool_name in LOCALIZATION_TOOLS:
        return "localization"

    if tool_name in EDITING_TOOLS:
        return "editing"

    if tool_name == "Bash":
        cmd = action.get("args", {}).get("command", "").lower()
        for pattern in SUBMISSION_PATTERNS:
            if pattern in cmd:
                return "submission"
        for pattern in VERIFICATION_PATTERNS:
            if pattern in cmd:
                return "verification"
        return "editing"

    return "editing"


def tag_phases_v2(steps: list[dict]) -> list[dict]:
    """Tag phases for a list of v2 steps. Returns phase boundary dicts."""
    if not steps:
        return []

    labels = [classify_action_v2(s) for s in steps]

    # Resolve inherited labels
    for i, label in enumerate(labels):
        if label == "inherited":
            labels[i] = labels[i - 1] if i > 0 else "editing"

    phases = []
    current_label = labels[0]
    start_step = steps[0]["step_id"]

    for i in range(1, len(labels)):
        if labels[i] != current_label:
            en, zh = PHASE_NAMES.get(current_label, (current_label.title(), current_label))
            phases.append({
                "name": en,
                "name_zh": zh,
                "start_step": start_step,
                "end_step": steps[i - 1]["step_id"],
                "label": current_label,
            })
            current_label = labels[i]
            start_step = steps[i]["step_id"]

    en, zh = PHASE_NAMES.get(current_label, (current_label.title(), current_label))
    phases.append({
        "name": en,
        "name_zh": zh,
        "start_step": start_step,
        "end_step": steps[-1]["step_id"],
        "label": current_label,
    })

    return phases


def detect_markers_v2(steps: list[dict]) -> list[dict]:
    """Detect behavioral markers (drift, churn, search-loop) in v2 steps."""
    markers = []
    n = len(steps)
    if n == 0:
        return markers

    markers.append({"step": steps[0]["step_id"], "type": "milestone", "reason": "Session start"})
    if n > 1:
        markers.append({"step": steps[-1]["step_id"], "type": "milestone", "reason": "Session end"})

    labels = [classify_action_v2(s) for s in steps]
    for i, label in enumerate(labels):
        if label == "inherited":
            labels[i] = labels[i - 1] if i > 0 else "editing"

    # Drift detection: 4+ consecutive edits without verification
    consecutive_edit = 0
    drift_marked = False
    for i, label in enumerate(labels):
        if label == "editing":
            consecutive_edit += 1
            if consecutive_edit >= 4 and not drift_marked:
                markers.append({
                    "step": steps[i]["step_id"],
                    "type": "drift-start",
                    "reason": f"{consecutive_edit} consecutive edits without verification",
                })
                drift_marked = True
        elif label == "verification":
            consecutive_edit = 0
            drift_marked = False
        else:
            if label != "editing":
                consecutive_edit = 0
                drift_marked = False

    # Search loop detection: 4+ consecutive localization
    consecutive_search = 0
    search_marked = False
    for i, label in enumerate(labels):
        if label == "localization":
            consecutive_search += 1
            if consecutive_search >= 4 and not search_marked:
                markers.append({
                    "step": steps[i]["step_id"],
                    "type": "search-loop",
                    "reason": "Extended search without editing — possible localization failure",
                })
                search_marked = True
        else:
            consecutive_search = 0
            search_marked = False

    # Churn detection: 3+ edits to same file in 5-step window
    churn_files: set[str] = set()
    for i in range(n):
        window = steps[max(0, i - 4):i + 1]
        file_edits: dict[str, int] = {}
        for s in window:
            action = s.get("action", {})
            tool_name = action.get("tool_name", "")
            if tool_name in ("Edit", "Write"):
                fp = action.get("args", {}).get("file_path", "")
                if fp:
                    file_edits[fp] = file_edits.get(fp, 0) + 1
        for fp, count in file_edits.items():
            if count >= 3 and fp not in churn_files:
                churn_files.add(fp)
                markers.append({
                    "step": steps[i]["step_id"],
                    "type": "churn",
                    "reason": f"Repeated edits to {fp} — possible edit recovery failure",
                })

    markers.sort(key=lambda m: (m["step"], m["type"]))
    return markers
```

```python
def _is_system_message(text: str) -> bool:

def _extract_text_content(content: Any) -> Optional[str]:
    if content is None:
        return None
    if isinstance(content, str):
        return content.strip() if content.strip() else None
    if isinstance(content, list):
        parts = []
        for block in content:
            if isinstance(block, dict) and block.get("type") == "text":
                parts.append(block.get("text", ""))
        combined = "".join(parts)
        return combined if combined else None
    return None

def _extract_tool_calls(content: Any) -> list[dict]:

    if isinstance(content, list):

def _parse_ts(ts_str: str) -> Optional[datetime]:
    if not ts_str:
        return None
    ts_str = ts_str.replace("Z", "+00:00")
    try:
        return datetime.fromisoformat(ts_str)
    except ValueError:
        return None


def _delta_seconds(ts1: str, ts2: str) -> float:
    dt1 = _parse_ts(ts1)
    dt2 = _parse_ts(ts2)
    if dt1 and dt2:
        return abs((dt2 - dt1).total_seconds())
    return 0.0


def parse_jsonl(filepath: str) -> list[dict]:
    """Parse JSONL file into a list of message entries."""
    entries = []
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                entry = json.loads(line)
            except json.JSONDecodeError:
                continue
            if entry.get("type") in ("user", "assistant"):
                entries.append(entry)
    return entries


def build_messages(entries: list[dict]) -> list[dict]:
    """Build v2 messages array from JSONL entries.

    Returns alternating user messages and agent runs grouped by turn_id.
    """
    messages: list[dict] = []
    turn_id = 0
    current_steps: list[dict] = []
    step_counter = 0
    prev_ts = ""

    # Pre-scan tool results by tool_use_id
    tool_result_map: dict[str, str] = {}
    for entry in entries:
        if entry.get("type") == "user":
            content = entry.get("message", {}).get("content", "")
            if isinstance(content, list):
                for block in content:
                    if isinstance(block, dict) and block.get("type") == "tool_result":
                        tid = block.get("tool_use_id", "")
                        result_text = _extract_tool_result_text(block.get("content", ""))
                        if tid:
                            tool_result_map[tid] = result_text

    i = 0
    while i < len(entries):
        entry = entries[i]
        msg_type = entry.get("type")
        content = entry.get("message", {}).get("content", "")
        timestamp = entry.get("timestamp", "")

        if msg_type == "user":
            # Skip tool_result entries
            if isinstance(content, list):
                is_tool_result = any(
                    isinstance(b, dict) and b.get("type") == "tool_result"
                    for b in content
                )
                if is_tool_result:
                    i += 1
                    continue

            text = _extract_text_content(content)

            # Skip system messages
            if text and _is_system_message(text):
                i += 1
                continue

            # Flush previous turn's agent run
            if current_steps:
                run_summary = _build_run_summary(current_steps)
                messages.append({
                    "turn_id": turn_id,
                    "agent_run_id": f"run_t{turn_id}",
                    "steps": current_steps,
                    "run_summary": run_summary,
                })
                current_steps = []
                step_counter = 0

            # New user turn
            turn_id += 1
            messages.append({
                "turn_id": turn_id,
                "role": "user",
                "content": text or "",
                "timestamp": timestamp,
            })
            if not prev_ts:
                prev_ts = timestamp
            i += 1
            continue

        if msg_type == "assistant":
            msg = entry.get("message", {})
            thinking = _extract_thinking(content)
            text = _extract_text_content(content)
            tool_calls = _extract_tool_calls(content)
            usage = _extract_usage(msg)

            if tool_calls:
                for tc in tool_calls:
                    step_counter += 1
                    tool_id = tc.get("id", "")
                    observation_text = tool_result_map.pop(tool_id, "")

                    current_steps.append({
                        "step_id": step_counter,
                        "phase": "",  # filled by phase tagger later
                        "thinking": thinking,
                        "thought": text or "",
                        "action": {
                            "tool_name": tc.get("name", "unknown"),
                            "tool_use_id": tool_id,
                            "args": tc.get("input", {}),
                        },
                        "observation": {
                            "type": "tool_result",
                            "text": observation_text[:2000],
                            "exit_code": None,
                        },
                        "state": None,
                        "status": "ok",
                        "timestamp": timestamp,
                        "execution_time": round(_delta_seconds(prev_ts, timestamp), 1),
                        "usage": usage,
                    })
                    # Only attach thinking to the first tool call in a batch
                    thinking = None
                    prev_ts = timestamp
            elif text:
                step_counter += 1
                current_steps.append({
                    "step_id": step_counter,
                    "phase": "",
                    "thinking": thinking,
                    "thought": text,
                    "action": {
                        "tool_name": "respond_to_user",
                        "tool_use_id": None,
                        "args": {},
                    },
                    "observation": {
                        "type": "text_response",
                        "text": "",
                        "exit_code": None,
                    },
                    "state": None,
                    "status": "ok",
                    "timestamp": timestamp,
                    "execution_time": round(_delta_seconds(prev_ts, timestamp), 1),
                    "usage": usage,
                })
                prev_ts = timestamp

        i += 1

    # Flush final agent run
    if current_steps:
        run_summary = _build_run_summary(current_steps)
        messages.append({
            "turn_id": turn_id,
            "agent_run_id": f"run_t{turn_id}",
            "steps": current_steps,
            "run_summary": run_summary,
        })

    return messages


def _build_run_summary(steps: list[dict]) -> dict:
    edit_failures = sum(1 for s in steps if s["status"] == "error")

    total_output = sum(s.get("usage", {}).get("output_tokens", 0) for s in steps)
    return {
        "result": "completed",
        "total_steps": len(steps),
        "edit_failures": edit_failures,
        "total_input_tokens": total_input,
        "total_output_tokens": total_output,
    }

            step["phase"] = phase_label_map.get(step["step_id"], "editing")
    else:
        phases = []
        markers = []

    # Extract metadata
    model = ""
    version = ""
    cwd = ""
    git_branch = ""
    for e in entries:
        if not model:

            if m:

            if c:

        if not git_branch:
            gb = e.get("gitBranch")
            if gb:
                git_branch = gb

    timestamps = [e.get("timestamp", "") for e in entries if e.get("timestamp")]
    start_time = timestamps[0] if timestamps else ""
    end_time = timestamps[-1] if timestamps else ""
    duration = _delta_seconds(start_time, end_time)

    # First user prompt
```

```python
    def _load_summary(self, path: Path, project: str) -> Optional[SessionSummary]:
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
        except (json.JSONDecodeError, OSError):
            return None

        summary = data.get("summary", {})
        if summary.get("status") == "empty":
            return None

        meta = data.get("session_metadata", {})
        self._mtime_cache[meta.get("session_id", "")] = path.stat().st_mtime

        return SessionSummary(
            conversation_id=data.get("conversation_id", ""),
            project=project,
            project_name=PROJECT_NAMES.get(project, project),
            status=summary.get("status", "unknown"),
            started_at=meta.get("started_at", ""),
            ended_at=meta.get("ended_at", ""),
            duration_sec=meta.get("duration_sec", 0),
            total_turns=summary.get("total_turns", 0),
            total_steps=summary.get("total_steps", 0),
            total_tool_calls=summary.get("total_tool_calls", 0),
            total_subagents=summary.get("total_subagents", 0),
            total_input_tokens=summary.get("total_input_tokens", 0),
            total_output_tokens=summary.get("total_output_tokens", 0),
            first_user_prompt=summary.get("first_user_prompt", ""),
            phase_labels=summary.get("phase_labels", []),
            marker_count=summary.get("marker_count", 0),
            exit_reason=summary.get("exit_reason", ""),
        )

    def list_sessions(
        self, project: Optional[str] = None, q: Optional[str] = None,
    ) -> list[SessionSummary]:
        result = self._index
        if project:
            result = [s for s in result if s.project == project]
        if q:
            q_lower = q.lower()
            result = [s for s in result if q_lower in s.first_user_prompt.lower()]
        return sorted(result, key=lambda s: s.started_at, reverse=True)

    def get_detail(self, project: str, session_hash: str) -> Optional[SessionDetail]:
        traj_path = self._data_dir / project / f"{session_hash}.traj"
        if not traj_path.exists():
            return None

        current_mtime = traj_path.stat().st_mtime

        # Find matching session_id
        session_id = None
        for s in self._index:
            if s.conversation_id.startswith(session_hash) and s.project == project:
                session_id = s.conversation_id
                break
        if not session_id:
            session_id = session_hash

        cached_mtime = self._mtime_cache.get(session_id, 0)
        if session_id in self._detail_cache and current_mtime <= cached_mtime:
            return self._detail_cache[session_id]

        try:
            with open(traj_path, "r", encoding="utf-8") as f:
                data = json.load(f)
        except (json.JSONDecodeError, OSError):
            return None

        summary_data = data.get("summary", {})
        meta = data.get("session_metadata", {})
        phases_raw = data.get("phases", [])
        markers_raw = data.get("markers", [])
        messages_raw = data.get("messages", [])
        subagents_raw = data.get("subagents", [])

        # Build marker lookup by step
        marker_map: dict[int, dict] = {}
        for m in markers_raw:
            marker_map.setdefault(m["step"], m)

        # Parse messages into typed objects
        parsed_messages: list = []
        for msg in messages_raw:
            if msg.get("role") == "user":
                parsed_messages.append(UserMessage(**{
                    k: msg[k] for k in ("turn_id", "role", "content", "timestamp")
                }))
            elif "steps" in msg:
                steps = []
                for s in msg["steps"]:
                    marker = marker_map.get(s["step_id"])
                    steps.append(StepDetail(
                        step_id=s["step_id"],
                        phase=s.get("phase", "editing"),
                        thinking=s.get("thinking"),
                        thought=s.get("thought", ""),
                        action=ActionDetail(**s["action"]),
                        observation=ObservationDetail(**s["observation"]),
                        state=s.get("state"),
                        status=s.get("status", "ok"),
                        timestamp=s.get("timestamp", ""),
                        execution_time=s.get("execution_time", 0),
                        usage=TokenUsage(**s["usage"]) if s.get("usage") else None,
                        marker=marker["type"] if marker else None,
                        marker_reason=marker["reason"] if marker else None,
                    ))
                parsed_messages.append(AgentRun(
                    turn_id=msg["turn_id"],
                    agent_run_id=msg.get("agent_run_id", ""),
                    steps=steps,
                    run_summary=RunSummary(**msg["run_summary"]),
                ))

        # Parse subagents
        subagent_summaries = [
            SubagentSummary(
                agent_id=sa["agent_id"],
                total_steps=sa.get("summary", {}).get("total_steps", 0),
                total_tool_calls=sa.get("summary", {}).get("total_tool_calls", 0),
                status=sa.get("summary", {}).get("status", "unknown"),
            )
            for sa in subagents_raw
        ]

        detail = SessionDetail(
            conversation_id=data.get("conversation_id", ""),
            project=project,
            project_name=PROJECT_NAMES.get(project, project),
            status=summary_data.get("status", "unknown"),
            started_at=meta.get("started_at", ""),
            ended_at=meta.get("ended_at", ""),
            duration_sec=meta.get("duration_sec", 0),
            total_turns=summary_data.get("total_turns", 0),
            total_steps=summary_data.get("total_steps", 0),
            total_tool_calls=summary_data.get("total_tool_calls", 0),
            total_subagents=summary_data.get("total_subagents", 0),
            total_input_tokens=summary_data.get("total_input_tokens", 0),
            total_output_tokens=summary_data.get("total_output_tokens", 0),
            first_user_prompt=summary_data.get("first_user_prompt", ""),
            phase_labels=summary_data.get("phase_labels", []),
            marker_count=summary_data.get("marker_count", 0),
            exit_reason=summary_data.get("exit_reason", ""),
            messages=parsed_messages,
            phases=[Phase(**p) for p in phases_raw],
            markers=[Marker(**m) for m in markers_raw],
            subagents=subagent_summaries,
        )

        self._detail_cache[session_id] = detail
        self._mtime_cache[session_id] = current_mtime
        return detail

    def list_projects(self) -> list[ProjectSummary]:
        projects: dict[str, list[SessionSummary]] = {}
        for s in self._index:
            projects.setdefault(s.project, []).append(s)

        result = []
        for proj, sessions in sorted(projects.items()):
            dates = [s.started_at[:10] for s in sessions if s.started_at]
            date_range = f"{min(dates)} ~ {max(dates)}" if dates else ""
            result.append(ProjectSummary(
                project=proj,
                project_name=PROJECT_NAMES.get(proj, proj),
                total_sessions=len(sessions),
                total_turns=sum(s.total_turns for s in sessions),
                total_steps=sum(s.total_steps for s in sessions),
                total_tool_calls=sum(s.total_tool_calls for s in sessions),
                total_input_tokens=sum(s.total_input_tokens for s in sessions),
                total_output_tokens=sum(s.total_output_tokens for s in sessions),
                date_range=date_range,
            ))
        return result

    def tool_stats(self) -> list[ToolStat]:
        tool_counts: dict[str, int] = {}
        for traj_path in self._data_dir.rglob("*.traj"):
            try:
                with open(traj_path) as f:
                    data = json.load(f)
                for msg in data.get("messages", []):
                    if "steps" in msg:
                        for step in msg["steps"]:
                            tool = step.get("action", {}).get("tool_name", "unknown")
                            if tool != "respond_to_user":
                                tool_counts[tool] = tool_counts.get(tool, 0) + 1
            except (json.JSONDecodeError, OSError):
                continue
        return sorted(
            [ToolStat(tool_name=k, count=v) for k, v in tool_counts.items()],
            key=lambda x: x.count,
            reverse=True,
        )

    def phase_stats(self) -> list[PhaseStat]:
        phase_turns: dict[str, int] = {}
        phase_sessions: dict[str, int] = {}
        for s in self._index:
            seen = set()
            for label in s.phase_labels:
                if label not in seen:
                    phase_sessions[label] = phase_sessions.get(label, 0) + 1
                    seen.add(label)
            for label in s.phase_labels:
                phase_turns[label] = phase_turns.get(label, 0) + 1
        result = []
        for phase, turns in phase_turns.items():
            sessions = phase_sessions.get(phase, 1)
            result.append(PhaseStat(
                phase=phase,
                total_turns=turns,
                avg_turns_per_session=round(turns / sessions, 1),
            ))
        return sorted(result, key=lambda x: x.total_turns, reverse=True)

    def token_stats(self) -> list[TokenStat]:
        projects: dict[str, dict] = {}
        for s in self._index:
            if s.project not in projects:
                projects[s.project] = {
                    "input": 0, "output": 0, "cache_read": 0, "cache_create": 0,
                }
            # Token totals per project come from per-session summaries
            projects[s.project]["input"] += s.total_input_tokens
            projects[s.project]["output"] += s.total_output_tokens

        # For cache stats, scan files
        for traj_path in self._data_dir.rglob("*.traj"):
            try:
                with open(traj_path) as f:
                    data = json.load(f)
                proj = data.get("session_metadata", {}).get("project", "")
                summary = data.get("summary", {})
                if proj in projects:
                    projects[proj]["cache_read"] += summary.get("total_cache_read_tokens", 0)
                    projects[proj]["cache_create"] += summary.get("total_cache_creation_tokens", 0)
            except (json.JSONDecodeError, OSError):
                continue

        result = []
        for proj, stats in sorted(projects.items()):
            total_input = stats["input"]
            cache_read = stats["cache_read"]
            cache_hit_rate = round(cache_read / total_input, 3) if total_input > 0 else 0.0
            result.append(TokenStat(
                project=proj,
                total_input_tokens=total_input,
                total_output_tokens=stats["output"],
                total_cache_read_tokens=cache_read,
                total_cache_creation_tokens=stats["cache_create"],
                cache_hit_rate=cache_hit_rate,
            ))
        return result
```

```typescript
const BASE = '/api'

async function fetchJson<T>(url: string): Promise<T> {
  const res = await fetch(url)
  if (!res.ok) {
    throw new Error(`API error: ${res.status} ${res.statusText}`)
  }
  return res.json() as Promise<T>
}

export function listTrajectories(
  project?: string,
  q?: string,
): Promise<SessionSummary[]> {
  const params = new URLSearchParams()
  if (project) params.set('project', project)
  if (q) params.set('q', q)
  const qs = params.toString()
  return fetchJson<SessionSummary[]>(`${BASE}/trajectories${qs ? '?' + qs : ''}`)
}

export function getTrajectory(
  project: string,
  sessionHash: string,
): Promise<SessionDetail> {
  return fetchJson<SessionDetail>(
    `${BASE}/trajectories/${encodeURIComponent(project)}/${encodeURIComponent(sessionHash)}`,
  )
}

export function listProjects(): Promise<ProjectSummary[]> {
  return fetchJson<ProjectSummary[]>(`${BASE}/projects`)
```
